# SETU — pinned dependencies (Python 3.11)

# Versions marked "PINNED" were specified by the project owner and must not be
# changed without an explicit decision. Everything else tracks the latest

# ---------- Agent framework ----------
langgraph==1.2.9                  # PINNED
langchain==1.3.11                 # PINNED
langchain-community                # RAGAS eval shim (eval/ only, see ragas_eval.py)
langchain-text-splitters          # RecursiveCharacterTextSplitter (parent chunks, ingestion/)
langchain-groq                    # primary LLM provider
langchain-ollama                  # local fallback provider
langchain-huggingface             # RAGAS's own context-similarity embeddings (eval/ only —
                                  # the runtime pipeline's embeddings are a Jina API call)

# ---------- Vector store & retrieval ----------
qdrant-client>=1.9,<2             # 1.x line; hybrid/native fusion API
fastembed                         # supplies the Qdrant/bm25 sparse model. Without it,
                                  # models.Document BM25 inference raises at query time.
sentence-transformers             # transitive need of langchain-huggingface (eval/ only —
                                  # BGE-M3 and the reranker themselves are Jina API calls)

# ---------- Ingestion ----------
docling                           # PDF -> structured markdown (primary)
crawl4ai                          # gov portal crawling; needs a one-time
                                  # `python -m playwright install chromium`
pymupdf4llm                       # lightweight PDF fallback when Docling is overkill

# ---------- Indic language layer ----------
# fasttext-wheel, not fasttext: it ships prebuilt Windows wheels, so no C++
# compiler is required. IndicLID's two classifiers are plain fastText models,
# kept local because they're tiny — translation itself is an Azure API call.
fasttext-wheel

# ---------- Serving ----------
fastapi
uvicorn[standard]
pydantic-settings                 # type-safe config from .env
httpx                             # Jina, Azure Translator, and Groq calls
chainlit>=2.11,<3                 # production UI; ui/requirements.txt has the
                                  # lean version actually shipped in its Docker image

# ---------- Evaluation ----------
ragas
deepeval

# ---------- Observability ----------
langfuse>=4,<5                    # v4 SDK; Langfuse Cloud or self-hosted, either works
